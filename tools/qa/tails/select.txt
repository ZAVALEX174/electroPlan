  // --- Б5, ч.1: ГРУППОВОЕ ВЫДЕЛЕНИЕ ПОСТОВ --------------------------------------------------------
  // На холсте уже 1 пост (post4 в кухне). Доставим ещё 3 поста в пустую зону НАД комнатами и проверим:
  // рамка по пустому месту выделяет посты, чьи ЦЕНТРЫ внутри; Ctrl+клик переключает член набора;
  // простой клик по пустому снимает; при повороте ВСЕГО плана на 90° рамка берёт те же посты, что
  // видны внутри неё на экране. Числа и скриншоты — в отчёт.
  n = 50;
  const cntPosts = () => ev(`return document.querySelectorAll(".plan-icon.post").length`);
  const cntSel   = () => ev(`return document.querySelectorAll(".plan-icon.post.selected").length`);
  const selText  = () => ev(`const e=document.getElementById("selectionCount");return e.hidden?"(скрыто)":e.textContent`);
  const postInfo = () => ev(`return Array.from(document.querySelectorAll(".plan-icon.post")).map(e=>{const r=e.getBoundingClientRect();return {id:e.dataset.id,num:e.textContent,cx:Math.round(r.left+r.width/2),cy:Math.round(r.top+r.height/2),sel:e.classList.contains("selected")}})`);
  const clip = async () => { const r = await rect(`document.querySelector(".canvas-scroll")`); return { x: r.x, y: r.y, width: r.w, height: r.h }; };
  const ctrlClickAt = async (x, y) => {
    await mouse(x, y, "mouseMoved");
    await c.send("Input.dispatchMouseEvent", { type: "mousePressed",  x, y, button: "left", clickCount: 1, modifiers: 2 });
    await c.send("Input.dispatchMouseEvent", { type: "mouseReleased", x, y, button: "left", clickCount: 1, modifiers: 2 });
    await sleep(400);
  };
  const place = async (fx, fy) => {
    await click(`document.querySelector("[data-place-template]")`); await sleep(300);
    const q = await canvasPt(fx, fy); await clickAt(q.x, q.y); await sleep(400);
  };

  // сброс режима/выделения и три поста в пустой зоне над комнатами (fy 0.12 — выше обеих комнат)
  await ev(`document.dispatchEvent(new KeyboardEvent("keydown",{key:"Escape",bubbles:true}));return 1;`); await sleep(200);
  await place(0.20, 0.12);
  await place(0.33, 0.12);
  await place(0.46, 0.12);
  log.push("Б5: постов на плане " + await cntPosts());
  await shot("sel-setup", [], await clip());

  // (1) РАМКА по пустому месту вокруг трёх верхних постов
  let aa = await canvasPt(0.14, 0.06), bb = await canvasPt(0.52, 0.18);
  await drag(aa.x, aa.y, bb.x - aa.x, bb.y - aa.y);
  log.push("(1) рамка: .selected=" + await cntSel() + ", надпись=«" + await selText() + "»");
  log.push("(1) посты: " + JSON.stringify(await postInfo()));
  await shot("sel-rubber", [], await clip());

  // (2) Ctrl+клик по выделенному посту → убрать из набора; ещё раз → вернуть
  let info = await postInfo();
  let selOne = info.find(p => p.sel);
  if (selOne) {
    await ctrlClickAt(selOne.cx, selOne.cy);
    log.push("(2) Ctrl+клик убрал пост №" + selOne.num + ": .selected=" + await cntSel() + ", надпись=«" + await selText() + "»");
    await shot("sel-ctrl-remove", [], await clip());
    await ctrlClickAt(selOne.cx, selOne.cy);
    log.push("(2) Ctrl+клик вернул: .selected=" + await cntSel() + ", надпись=«" + await selText() + "»");
    await shot("sel-ctrl-add", [], await clip());
  } else log.push("(2) ПРОПУСК: рамка никого не выделила");

  // (3) простой клик по пустому месту снимает выделение
  let emp = await canvasPt(0.14, 0.45);
  await clickAt(emp.x, emp.y);
  log.push("(3) клик по пустому: .selected=" + await cntSel() + ", надпись=«" + await selText() + "»");

  // (4) поворот ВСЕГО плана на 90° и рамка — берёт те же посты, что видны внутри неё на экране
  await setVal(`document.getElementById("rotateTargetSelect")`, "world");
  await click(`document.getElementById("planRotateRightBtn")`); await sleep(700);
  log.push("(4) transform холста: " + await ev(`return getComputedStyle(document.getElementById("canvas")).transform`));
  let before = await postInfo();
  log.push("(4) экранные центры постов после поворота: " + JSON.stringify(before.map(p => ({ num: p.num, cx: p.cx, cy: p.cy }))));
  // ТЕСНУЮ рамку строим вокруг экранных центров двух постов с наименьшим cy (соседних по вертикали),
  // чтобы показать РАЗЛИЧЕНИЕ: прочие посты должны остаться невыделенными. Старт — на 45px выше-левее
  // верхнего центра (заведомо пустое место, не иконка).
  let sorted = before.slice().sort((p, q) => p.cy - q.cy);
  let pick = sorted.slice(0, 2);
  let minx = Math.min(...pick.map(p => p.cx)) - 45, maxx = Math.max(...pick.map(p => p.cx)) + 28;
  let miny = Math.min(...pick.map(p => p.cy)) - 45, maxy = Math.max(...pick.map(p => p.cy)) + 28;
  await drag(minx, miny, maxx - minx, maxy - miny);
  let after = await postInfo();
  log.push("(4) рамка при 90°: ждём внутри " + JSON.stringify(pick.map(p => p.num).sort())
    + ", выделены " + JSON.stringify(after.filter(p => p.sel).map(p => p.num).sort())
    + ", .selected=" + await cntSel() + ", надпись=«" + await selText() + "»");
  await shot("sel-rubber-rot90", [], await clip());

  // --- Б5, ч.2: УДАЛЕНИЕ и СДВИГ СТРЕЛКАМИ выделенной ГРУППЫ ----------------------------------------
  // (5) рамкой выделить посты над комнатами → сумма сметы → Delete → постов/суммы нет → Ctrl+Z → все
  //     вернулись, сумма как до (ОДИН шаг). (6) два поста у границы комнаты → удержание стрелки (CDP
  //     autoRepeat, как в undo.txt) переводит их в соседнюю комнату → roomId в сохранённом проекте
  //     сменился и сумма пересчитана → один Ctrl+Z → всё как до. Числа и скриншоты — в отчёт.
  n = 60;
  // мир после сцены (4) повёрнут на 90° — возвращаем ОДНИМ поворотом влево (90°→0°), чтобы состояние
  // вида (угол И pan) точно совпало с исходным: доворот тремя вправо (до 360°) оставлял вид в состоянии,
  // где попадание рамки расходилось с отрисовкой и рамка никого не брала.
  await click(`document.getElementById("planRotateLeftBtn")`); await sleep(700);
  await ev(`document.dispatchEvent(new KeyboardEvent("keydown",{key:"Escape",bubbles:true}));return 1;`); await sleep(200);
  log.push("(5-6) мир возвращён к 0°: transform=" + await ev(`return getComputedStyle(document.getElementById("canvas")).transform`));

  const grand = () => ev(`const e=document.getElementById("grandTotal")||document.querySelector(".grand-total");return e?e.textContent.trim():null`);
  // настоящее нажатие Ctrl+Z через CDP (modifiers:2=Ctrl) — как в undo.txt
  const hotkey = async (code, vk, keyChar) => {
    await c.send("Input.dispatchKeyEvent", { type: "keyDown", modifiers: 2, code, key: keyChar, windowsVirtualKeyCode: vk, nativeVirtualKeyCode: vk });
    await c.send("Input.dispatchKeyEvent", { type: "keyUp",   modifiers: 2, code, key: keyChar, windowsVirtualKeyCode: vk, nativeVirtualKeyCode: vk });
    await sleep(600);
  };
  const ctrlZ = () => hotkey("KeyZ", 90, "z");
  const delKey = async () => { await ev(`document.activeElement&&document.activeElement.blur&&document.activeElement.blur();document.dispatchEvent(new KeyboardEvent("keydown",{key:"Delete",bubbles:true}));return 1;`); await sleep(600); };
  const roomOf = ids => ev(`const want=${JSON.stringify(ids)};return (JSON.parse(localStorage.getItem(ProjectStore.key)||"{}").posts||[]).filter(p=>want.includes(p.id)).map(p=>({id:p.id,room:p.roomId||null,x:Math.round(p.x)}))`);
  const roomRects = () => ev(`return Array.from(document.querySelectorAll(".room-label")).map(e=>{const r=e.getBoundingClientRect();return {name:e.textContent.trim(),cx:Math.round(r.left+r.width/2),cy:Math.round(r.top+r.height/2)}})`);
  const placeAt = async (x, y) => { await click(`document.querySelector("[data-place-template]")`); await sleep(300); await clickAt(x, y); await sleep(500); };

  // (5) группу из верхнего ряда постов формируем ТОЧЕЧНО (клик + Ctrl+клик по центрам иконок): после
  // цикла поворотов мира в сцене (4) попадание рамки расходится с отрисовкой (hit-test рамки берёт пост
  // в комнате, но не верхний ряд, хотя их центры внутри — см. отчёт). Выделение РАМКОЙ доказано сценой
  // (1); задача сцены (5) — проверить УДАЛЕНИЕ группы и возврат Ctrl+Z, для чего состав группы надёжнее
  // собрать прямыми кликами (как в сцене 6 — попадают по реальным узлам независимо от состояния вида).
  const before5 = await postInfo();
  log.push("(5) посты на холсте: " + JSON.stringify(before5.map(p => ({ num: p.num, cx: p.cx, cy: p.cy }))));
  const sorted5 = before5.slice().sort((p, q) => p.cy - q.cy);
  const pick5 = sorted5.filter(p => p.cy <= sorted5[0].cy + 60);   // верхний ряд постов
  await clickAt(pick5[0].cx, pick5[0].cy);                          // одиночное выделение первого
  for (let i = 1; i < pick5.length; i++) await ctrlClickAt(pick5[i].cx, pick5[i].cy);   // Ctrl+клик добавляет остальных
  const sum5b = await grand(), posts5b = await cntPosts(), sel5 = await cntSel();
  log.push("(5) группа собрана кликами: .selected=" + sel5 + ", надпись=«" + await selText() + "», сумма=" + sum5b + ", постов=" + posts5b);
  await shot("del-group-selected", [`document.querySelector(".grand-total")`], await clip());
  await delKey();
  const posts5a = await cntPosts(), sum5a = await grand();
  log.push("(5) Delete группы: постов " + posts5b + "→" + posts5a + " (ждём −" + sel5 + "), сумма " + sum5b + "→" + sum5a + ", надпись=«" + await selText() + "»");
  await shot("del-group-after", [`document.querySelector(".grand-total")`], await clip());
  await ctrlZ();
  const posts5u = await cntPosts(), sum5u = await grand();
  log.push("(5) Ctrl+Z: постов " + posts5a + "→" + posts5u + " (ждём " + posts5b + "), сумма " + sum5a + "→" + sum5u + " (ждём " + sum5b + ") — один шаг вернул всё");
  await shot("del-group-undo", [`document.querySelector(".grand-total")`], await clip());

  // (6) два поста у границы комнаты A → удержание стрелки переводит их в комнату B
  n = 70;
  const rooms6 = await roomRects();
  log.push("(6) комнаты (экранные центры подписей): " + JSON.stringify(rooms6));
  if (rooms6.length >= 2) {
    const A = rooms6[0], B = rooms6[1];
    // точка внутри A, на 30% пути к центру B (у границы, но ещё в A); два поста рядом
    const px = Math.round(A.cx + (B.cx - A.cx) * 0.30), py = Math.round(A.cy + (B.cy - A.cy) * 0.30);
    // два НОВЫХ поста у границы A; опознаём их по разнице списка id до/после (рамка могла бы зацепить
    // соседний пост комнаты — поэтому выделяем ровно эти два точечно: клик + Ctrl+клик по их центрам)
    const idsBefore6 = await ev(`return Array.from(document.querySelectorAll(".plan-icon.post")).map(e=>e.dataset.id)`);
    await placeAt(px - 14, py);
    await placeAt(px + 14, py);
    const idsAfter6 = await ev(`return Array.from(document.querySelectorAll(".plan-icon.post")).map(e=>e.dataset.id)`);
    const newIds = idsAfter6.filter(id => !idsBefore6.includes(id));
    const info6 = await postInfo();
    const n1 = info6.find(p => p.id === newIds[0]), n2 = info6.find(p => p.id === newIds[1]);
    await clickAt(n1.cx, n1.cy);          // одиночное выделение первого нового поста
    await ctrlClickAt(n2.cx, n2.cy);      // Ctrl+клик добавляет второй → группа из двух
    const selIds = await ev(`return Array.from(document.querySelectorAll(".plan-icon.post.selected")).map(e=>e.dataset.id)`);
    log.push("(6) два новых поста " + JSON.stringify(newIds) + " у границы «" + A.name + "»; выделено " + JSON.stringify(selIds) + ", .selected=" + await cntSel() + ", надпись=«" + await selText() + "»");
    const before6 = await roomOf(selIds), sum6b = await grand();
    log.push("(6) ДО сдвига: " + JSON.stringify(before6) + ", сумма=" + sum6b);
    await shot("move-group-before", [`document.querySelector(".grand-total")`], await clip());
    // направление A→B по доминирующей оси; удержание (autoRepeat) ~90 повторов = один шаг истории
    const dX = B.cx - A.cx, dY = B.cy - A.cy, horiz = Math.abs(dX) >= Math.abs(dY);
    const key = horiz ? (dX > 0 ? "ArrowRight" : "ArrowLeft") : (dY > 0 ? "ArrowDown" : "ArrowUp");
    const vk = { ArrowRight: 39, ArrowLeft: 37, ArrowDown: 40, ArrowUp: 38 }[key];
    log.push("(6) удержание клавиши " + key + " (перенос в сторону комнаты «" + B.name + "»)");
    for (let i = 0; i < 90; i++) await c.send("Input.dispatchKeyEvent", { type: "rawKeyDown", code: key, key, windowsVirtualKeyCode: vk, nativeVirtualKeyCode: vk, autoRepeat: i > 0 });
    await c.send("Input.dispatchKeyEvent", { type: "keyUp", code: key, key, windowsVirtualKeyCode: vk, nativeVirtualKeyCode: vk });
    await sleep(900);
    const after6 = await roomOf(selIds), sum6a = await grand();
    log.push("(6) ПОСЛЕ удержания: " + JSON.stringify(after6) + ", сумма=" + sum6a + " (ждём: roomId у постов сменился с комнаты A на B, сумма пересчитана)");
    await shot("move-group-after", [`document.querySelector(".grand-total")`], await clip());
    await ctrlZ();
    const undo6 = await roomOf(selIds), sum6u = await grand();
    log.push("(6) один Ctrl+Z: " + JSON.stringify(undo6) + ", сумма=" + sum6u + " (ждём = ДО: " + JSON.stringify(before6) + ", сумма " + sum6b + ")");
    await shot("move-group-undo", [`document.querySelector(".grand-total")`], await clip());
  } else log.push("(6) ПРОПУСК: на холсте меньше двух комнат с подписями");
