  // ===================================================================================================
  // Б9 «КОПИРОВАНИЕ ПОСТОВ», браузерный хвост `copy`. Ctrl+C/Ctrl+V/Ctrl+Z/Ctrl+Y и движение мыши —
  // НАСТОЯЩИЕ события через DevTools Protocol (Input.dispatchKeyEvent / Input.dispatchMouseEvent), чтобы
  // проверить проводку клавиш (document.onkeydown → EPPostCopyUi.pastePosts) и трекер «мышь над планом»
  // (canvasInput.js canvasPointer). Вызовов copyPosts()/pastePosts() из консоли НЕТ.
  //
  // ⚠️ app.js — IIFE: ни state, ни product/buildEstimate/clientToWorld из консоли не видны. Поэтому
  // читаем ТОЛЬКО через DOM, localStorage (ProjectStore), EP_DATA и глобальные EP*-модули — как select.txt.
  //
  // Фраза владельца (критерий): «Выделили посты в одном номере, Ctrl+C, навели мышь на соседний номер,
  // Ctrl+V — там такие же посты в том же взаимном расположении; каждый считается по своей комнате, смета
  // растёт на их стоимость; Ctrl+Z убирает вставленное одним шагом». Решения владельца 10.10: копии несут
  // СВОИ новые номера проходных (своя пара, с исходными не связываются); накладка копии — отделки комнаты;
  // подходящая накладка НЕ меняется; нет подходящей — остальные вставить, про невставленный сказать;
  // имя группы света у копии, вставленной туда же, где исходный, снимается с сообщением-крестиком.
  // Все деньги — в ЕВРО прайса (currencySelect=EUR): тогда «до копейки» проверяемо без ошибки курса.
  //
  // Эталон поста (из независимой проверки 11.10, RESULT.md): накладка 21653.01 (Eikon Evo, 3М, 68,07 €) +
  // 3×клавиши 20021.B (5,08 €). Номер проходной «7» у первой клавиши → пост 157,41 €, пара 314,82 €,
  // четыре поста (пара + две копии своей пары) 629,64 €.
  //
  // Проверки — с ПАДЕНИЕМ (assertOk валит сценарий исключением → harness печатает «ОШИБКА» и exit 1).
  // ===================================================================================================
  n = 100;
  const assertOk = (cond, msg) => { log.push((cond ? "ОК" : "FAIL") + ": " + msg); if (!cond) throw new Error(msg); };
  const settle = () => sleep(900);   /* автосохранение в localStorage дебаунсится на 700 мс (EPConfig) */

  // Настоящие клавиши через CDP (modifiers:2 = Ctrl), ровно как в tails/select.txt.
  const hotkey = async (code, vk, keyChar, mods = 2) => {
    await c.send("Input.dispatchKeyEvent", { type: "keyDown", modifiers: mods, code, key: keyChar, windowsVirtualKeyCode: vk, nativeVirtualKeyCode: vk });
    await c.send("Input.dispatchKeyEvent", { type: "keyUp",   modifiers: mods, code, key: keyChar, windowsVirtualKeyCode: vk, nativeVirtualKeyCode: vk });
    await sleep(500);
  };
  const ctrlC = () => hotkey("KeyC", 67, "c");
  const ctrlV = () => hotkey("KeyV", 86, "v");
  const ctrlZ = () => hotkey("KeyZ", 90, "z");
  const ctrlY = () => hotkey("KeyY", 89, "y");
  const ctrlClickAt = async (x, y) => {
    await mouse(x, y, "mouseMoved");
    await c.send("Input.dispatchMouseEvent", { type: "mousePressed",  x, y, button: "left", clickCount: 1, modifiers: 2 });
    await c.send("Input.dispatchMouseEvent", { type: "mouseReleased", x, y, button: "left", clickCount: 1, modifiers: 2 });
    await sleep(400);
  };
  // Навести мышь: НАСТОЯЩИЙ mouseMoved → pointermove на .canvas-scroll → lastPointer.overCanvas=true.
  const hoverAt = async (x, y) => { await mouse(x, y, "mouseMoved"); await sleep(250); };

  const clip = async () => { const r = await rect(`document.querySelector(".canvas-scroll")`); return { x: r.x, y: r.y, width: r.w, height: r.h }; };
  const cntPosts = () => ev(`return document.querySelectorAll(".plan-icon.post").length`);
  const cntSel   = () => ev(`return document.querySelectorAll(".plan-icon.post.selected").length`);
  const selText  = () => ev(`const e=document.getElementById("selectionCount");return e.hidden?"(скрыто)":e.textContent`);
  const selIds   = () => ev(`return Array.from(document.querySelectorAll(".plan-icon.post.selected")).map(function(e){return e.dataset.id;})`);
  const toastText = () => ev(`const e=document.querySelector(".toast");return e?e.textContent.trim():null`);
  const lightText = () => ev(`const e=document.getElementById("lightingSummary");return e?e.textContent.replace(/\\s+/g," ").trim():""`);
  // Сообщения с крестиком (EPNotices) — НЕ toast: живут в #notices как .notice, текст в .notice__text.
  const notices = () => ev(`return Array.from(document.querySelectorAll("#notices .notice")).map(function(e){var t=e.querySelector(".notice__text");return t?t.textContent.trim():e.textContent.trim();});`);
  // Деньги: в EUR формат ru-RU «1 234,56 €» — пробел-тысячи, запятая-дробь; парсим в число до копейки.
  const parseMoney = s => { if (s == null) return NaN; let t = String(s).replace(/[\s  ]/g, "").replace(/[^0-9.,-]/g, ""); if (t.indexOf(",") >= 0) t = t.replace(/\./g, "").replace(",", "."); return parseFloat(t); };
  const equip = async () => parseMoney(await ev(`return document.getElementById("equipmentTotal").textContent`));
  const grand = async () => parseMoney(await ev(`return document.getElementById("grandTotal").textContent`));
  const prodId = async (code) => ev(`var p=((window.EP_DATA&&window.EP_DATA.products)||[]).find(function(x){return x.code===${JSON.stringify(code)};});return p?Number(p.id):null;`);

  // Посты из DOM (экранные центры, выделение).
  const domPosts = () => ev(`return Array.from(document.querySelectorAll(".plan-icon.post")).map(function(e){var r=e.getBoundingClientRect();return {id:e.dataset.id,cx:Math.round(r.left+r.width/2),cy:Math.round(r.top+r.height/2),sel:e.classList.contains("selected")};});`);
  // Посты из сохранённого проекта (мировые x/y, комната, номер, накладка, механизмы, проходные).
  const prjPosts = () => ev(`var j=JSON.parse(localStorage.getItem(ProjectStore.key)||"{}");return (j.posts||[]).map(function(p){return {id:p.id,num:Number(p.number),room:p.roomId||null,x:p.x,y:p.y,frameId:p.frameId,mech:(p.mechanismIds||[]).slice(),cross:(p.keyCrossNumbers||[]).slice(),groups:(p.keyGroups||[]).slice()};});`);
  // Слить обе картины по id (мир + экран + выделение).
  const posts = async () => { const d = await domPosts(), p = await prjPosts(); return p.map(x => { const dd = d.find(y => y.id === x.id) || {}; return Object.assign({}, x, { cx: dd.cx ?? null, cy: dd.cy ?? null, sel: !!dd.sel }); }); };
  // Код накладки поста по frameId (для проверки «накладка копии = накладка исходного»).
  const frameCodeOf = (frameId) => ev(`var p=((window.EP_DATA&&window.EP_DATA.products)||[]).find(function(x){return String(x.id)===String(${JSON.stringify(frameId)});});return p?p.code:null;`);
  const roomsInfo = async () => {
    const p = await ev(`var j=JSON.parse(localStorage.getItem(ProjectStore.key)||"{}");return (j.rooms||[]).map(function(r){return {id:r.id,name:r.name};});`);
    const d = await ev(`return Array.from(document.querySelectorAll(".room-label")).map(function(e){var r=e.getBoundingClientRect();return {name:e.textContent.trim(),cx:Math.round(r.left+r.width/2),cy:Math.round(r.top+r.height/2)};});`);
    return p.map(r => { const dd = d.find(x => x.name.indexOf(r.name) >= 0) || {}; return { id: r.id, name: r.name, cx: dd.cx ?? null, cy: dd.cy ?? null }; });
  };
  // Инструмент «Выбор»: makeRoom оставляет активным «Комната», и в этом режиме mousedown холста гасит
  // всплытие к canvas.onclick (placePendingAtEvent) — размещение и выделение кликом не доходят. Возвращаем
  // «Выбор» перед любой работой мышью по холсту.
  const selectTool = async () => { await click(`document.querySelector('[data-tool="select"]')`); await sleep(200); };
  const placeLast = async (x, y) => {
    // Закрываем возможную модалку конструктора: пока postModal открыт, кнопка «Разместить» под ним и клик
    // по ней не доходит (режим размещения не включается). Esc в app.js закрывает окно поста.
    if (await ev(`return document.getElementById("postModal").classList.contains("open")`)) {
      await ev(`document.dispatchEvent(new KeyboardEvent("keydown",{key:"Escape",bubbles:true}));return 1;`); await sleep(300);
    }
    await selectTool();
    // Библиотека перерисовывается ПОСЛЕ savePost асинхронно: пока renderLibrary не отработал, кнопок
    // «Разместить» в DOM нет (замерено: до ~2 с список бывает пуст). Ждём появления кнопки, иначе клик
    // уходит в пустоту и режим размещения не включается. Это ожидание рендера, не «пауза на всякий случай».
    for (let i = 0; i < 25; i++) { if (await ev(`return document.querySelectorAll("[data-place-template]").length>0`)) break; await sleep(300); }
    // Взводим размещение последнего шаблона его же кнопкой «Разместить» (НЕ проверяемый шаг — проверяем
    // копирование). onclick кнопки идемпотентно добавляет класс "placing"; опрашиваем до включения режима.
    const arm = async () => { await ev(`var b=Array.from(document.querySelectorAll("[data-place-template]")).at(-1);if(b)b.click();return 1;`); await sleep(300); return ev(`return document.getElementById("canvas").classList.contains("placing")`); };
    let placing = false;
    for (let i = 0; i < 6 && !placing; i++) placing = await arm();
    await clickAt(x, y); await sleep(500);
    log.push("placeLast@(" + x + "," + y + "): placing=" + placing + ", шаблонов=" + await ev(`return document.querySelectorAll("[data-place-template]").length`) + ", постов=" + await cntPosts());
  };
  // Накатать чистый холст: настоящая кнопка «Очистить холст» (#clearBtn → clearCanvas, без подтверждения)
  // стирает посты И комнаты; валюту ставим ЕВРО (прайс), чтобы смета была в 2-знаковых евро без курса.
  // Сообщение «ничего не вставлено» живёт до крестика и переживает очистку холста (у него нет pruneWhen —
  // это не отмена). Между сценами снимаем такие висящие сообщения из DOM, чтобы они не мешали следующей
  // сцене считать свои сообщения. Это уборка стенда между сценами, а не обход проверки.
  const clearAll = async () => {
    await click($("clearBtn")); await sleep(500);
    await ev(`document.querySelectorAll("#notices .notice").forEach(function(e){e.remove()});return 1;`);
    await setVal($("currencySelect"), "EUR"); await sleep(400);
  };
  // Отделка комнаты: клик по подписи → селекты свойств (ids как в shoot.mjs). vals — {selectId: value}.
  const setRoomProps = async (label, vals) => {
    await selectTool();
    await clickAt(label.cx, label.cy); await sleep(500);
    for (const k of Object.keys(vals)) await setVal($(k), vals[k]);
    await sleep(500);
    await clickAt(10, 300); await sleep(200);   // снять выделение/закрыть свойства
  };
  // Построить шаблон поста: накладка frameCode + keyCount клавиш keyCode через НАСТОЯЩИЙ конструктор.
  // roomNameSub — подстрока имени комнаты для контекста builderRoomSelect; опцию ищем ПОСЛЕ открытия окна
  // (до открытия список опций пуст/устаревший). Контекст комнаты БЕЗ отделки гарантирует, что любая
  // накладка видна в списке и ставится. slotCount — число модулей накладки; задать ДО выбора накладки,
  // иначе накладка иной модульности (напр. 8М 2+2+2+2) не встанет под дефолтные 3 слота и каталог клавиш
  // под неё не построится (как в scn.mjs).
  const buildTemplate = async (frameCode, keyCode, keyCount, roomNameSub, slotCount) => {
    await click($("newPostBtn")); await sleep(800);
    if (roomNameSub) {
      const opt = await ev(`return (Array.from(document.getElementById("builderRoomSelect").options||[]).find(function(o){return o.textContent.indexOf(${JSON.stringify(roomNameSub)})>=0;})||{}).value`);
      if (opt) { await setVal($("builderRoomSelect"), opt); await sleep(300); }
    }
    if (slotCount) { await setVal($("postSlotCount"), String(slotCount)); await sleep(400); }
    const fId = await prodId(frameCode);
    assertOk(fId != null, "накладка " + frameCode + " есть в каталоге");
    await setVal($("postFrameSelect"), String(fId)); await sleep(500);
    const kId = await prodId(keyCode);
    await setVal($("builderSearch"), keyCode); await sleep(600);
    const kCard = `document.querySelector('#builderCatalog [data-pick="${kId}"]')`;
    assertOk(await ev(`return !!${kCard}`), "карточка клавиши " + keyCode + " доступна в конструкторе для накладки " + frameCode);
    for (let i = 0; i < keyCount; i++) { await click(kCard); await sleep(350); }
    const slots = await ev(`return JSON.stringify(Array.from(document.querySelectorAll("#builderSlots .slot-code")).map(function(e){return e.textContent;}))`);
    const dis = await ev(`return document.getElementById("savePost").disabled`);
    log.push("buildTemplate " + frameCode + "+" + keyCount + "×" + keyCode + ": слоты=" + slots + ", savePost.disabled=" + dis);
    await click($("savePost")); await sleep(700);
    return { frameCode };
  };
  // Открыть окно размещённого поста по его экранным координатам (единственное место полей «№ проходной»
  // и «Группа света»): выделить кликом, «Редактировать».
  const openPlacedPost = async (cx, cy) => {
    await selectTool();
    await clickAt(cx, cy); await sleep(300);
    await click(btnText("Редактировать")); await sleep(900);
  };
  // Вписать значение в поле слота окна поста (data-slot-cross / data-slot-group) по индексу клавиши.
  const setSlotField = async (attr, index, value) => {
    const n0 = await ev(`return document.querySelectorAll("#builderSlots [${attr}]").length`);
    await ev(`var i=document.querySelector('#builderSlots [${attr}="${index}"]');if(i){i.value=${JSON.stringify(String(value))};i.dispatchEvent(new Event("input",{bubbles:true}));i.dispatchEvent(new Event("change",{bubbles:true}));}return 1;`);
    await sleep(400);
    return n0;
  };

  // --- СЦЕНА 1. Пара постов с общим номером проходной в первой комнате → Ctrl+C → мышь в соседнюю → Ctrl+V.
  // Эталон: 21653.01 + 3×20021.B, проходная «7» у первой клавиши → 157,41 € / пост, пара 314,82 €.
  // Ждём: +2 поста, обе копии в соседней комнате, взаимное расположение = исходному, у копий СВОЙ общий
  // номер проходной (≠7), оборудование 629,64 €, выделены новые, гаснущее «Вставлено постов: 2.».
  await clearAll();
  await makeRoom(0.24, 0.46); await nameRoom("Номер 101");
  await makeRoom(0.52, 0.46); await nameRoom("Номер 102");
  await settle();
  let rooms1 = await roomsInfo();
  log.push("(1) SETUP комнаты: " + JSON.stringify(rooms1));
  assertOk(rooms1.length === 2, "(1) созданы РОВНО две комнаты");
  const r101 = rooms1.find(r => r.name.indexOf("101") >= 0), r102 = rooms1.find(r => r.name.indexOf("102") >= 0);

  await buildTemplate("21653.01", "20021.B", 3, "101");

  // Два поста в комнате 101 рядом (центр значка = x+12,y+12).
  await placeLast(r101.cx - 22, r101.cy + 45);
  await placeLast(r101.cx + 22, r101.cy + 45);
  await settle();
  log.push("(1) SETUP поставлено постов: " + await cntPosts());
  assertOk(await cntPosts() === 2, "(1) в комнате 101 поставлено два поста");

  // Фикстура «парная проходная»: каждому посту через окно поста задаём ОДИН И ТОТ ЖЕ номер «7» у первой
  // клавиши (slot 0). Это исходные данные владельца «одинаковый номер = пара», не проверяемый шаг.
  const setCross0 = async (cx, cy, num) => {
    await openPlacedPost(cx, cy);
    const has = await setSlotField("data-slot-cross", 0, num);
    await click($("savePost")); await sleep(700);
    return has;
  };
  let pp = await posts();
  const kiSet = p => p.cross.findIndex(v => String(v || "").trim() !== "");
  const hasField1 = await setCross0(pp[0].cx, pp[0].cy, "7");
  pp = await posts();
  const target2 = pp.find(p => kiSet(p) < 0) || pp[1];
  const hasField2 = await setCross0(target2.cx, target2.cy, "7");
  await settle();
  log.push("(1) SETUP полей «№ проходной» в окне поста: пост1=" + hasField1 + ", пост2=" + hasField2);
  assertOk(hasField1 >= 1 && hasField2 >= 1, "(1) у размещённого поста есть поле «№ проходной»");
  const srcSnap = (await posts()).sort((a, b) => a.num - b.num);
  log.push("(1) SETUP исходные посты: " + JSON.stringify(srcSnap.map(p => ({ num: p.num, room: p.room, x: +p.x.toFixed(1), y: +p.y.toFixed(1), cross: p.cross }))));
  assertOk(srcSnap.length === 2 && srcSnap.every(p => p.cross[kiSet(p)] === "7"), "(1) обоим постам задан один номер проходной «7» (пара)");

  const S0 = await equip(), N0 = await cntPosts();
  const light0 = await lightText();
  log.push("(1) SETUP S0(оборудование,€)=" + S0 + ", постов=" + N0 + "; группы света: " + light0);
  assertOk(Math.abs(S0 - 314.82) < 0.02, "(1) пара = 314,82 € (эталонный пост 21653.01+3×20021.B, проходная 7): факт " + S0);
  await ev(`document.dispatchEvent(new KeyboardEvent("keydown",{key:"Escape",bubbles:true}));return 1;`); await sleep(200);
  await clickAt(10, 300); await sleep(200);   // снять выделение
  await shot("copy-setup", [`document.querySelector(".grand-total")`], await clip());

  // Ctrl+C группы → мышь в комнату 102 → Ctrl+V
  let info2 = (await posts()).sort((a, b) => a.num - b.num);
  const srcA = info2[0], srcB = info2[1];
  await clickAt(srcA.cx, srcA.cy);
  await ctrlClickAt(srcB.cx, srcB.cy);
  log.push("(1) выделено перед Ctrl+C: .selected=" + await cntSel() + ", надпись=«" + await selText() + "», ids=" + JSON.stringify(await selIds()));
  assertOk(await cntSel() === 2, "(1) перед Ctrl+C выделены оба исходных поста");
  await ctrlC();

  // МИРОВАЯ точка под мышью: читаем ровно ту, что отдаст clientToWorld, через transform холста.
  const r102c = (await roomsInfo()).find(r => r.name.indexOf("102") >= 0);
  const worldPt = async (cx, cy) => ev(`
    var canvas=document.getElementById("canvas"); var s=document.querySelector(".canvas-scroll").getBoundingClientRect();
    var t=getComputedStyle(canvas).transform; var m=t&&t!=="none"?t.match(/matrix\\(([^)]+)\\)/):null;
    var a=1,b=0,cc=0,d=1,e=0,f=0; if(m){var v=m[1].split(",").map(Number);a=v[0];b=v[1];cc=v[2];d=v[3];e=v[4];f=v[5];}
    var px=${cx}-s.left-e, py=${cy}-s.top-f; var det=a*d-b*cc;
    return {x:(px*d-py*cc)/det, y:(-px*b+py*a)/det};
  `);
  await hoverAt(r102c.cx, r102c.cy);
  const expectPoint = await worldPt(r102c.cx, r102c.cy);
  log.push("(1) мышь в центр 102 (client " + r102c.cx + "," + r102c.cy + ") → мировая точка " + JSON.stringify({ x: +expectPoint.x.toFixed(2), y: +expectPoint.y.toFixed(2) }));
  await ctrlV();
  await settle();

  const after2 = await posts();
  const N2 = after2.length;
  log.push("(1) постов после Ctrl+V: " + N0 + "→" + N2 + ", тост=«" + await toastText() + "»");
  assertOk(N2 === N0 + 2, "(1) постов стало +2 (было " + N0 + ", стало " + N2 + ")");
  assertOk(/Вставлено постов: 2\./.test(await toastText() || ""), "(1) гаснущее «Вставлено постов: 2.»");

  const srcIds = new Set(srcSnap.map(p => p.id));
  const copies = after2.filter(p => !srcIds.has(p.id)).sort((a, b) => a.num - b.num);
  const srcAfter = after2.filter(p => srcIds.has(p.id)).sort((a, b) => a.num - b.num);
  assertOk(copies.length === 2, "(1) ровно две НОВЫЕ копии");

  log.push("(1) комнаты: исходные " + JSON.stringify(srcAfter.map(p => p.room)) + " (ждём 101=" + r101.id + "), копии " + JSON.stringify(copies.map(p => p.room)) + " (ждём 102=" + r102.id + ")");
  assertOk(srcAfter.every(p => p.room === r101.id), "(1) исходные остались в комнате 101");
  assertOk(copies.every(p => p.room === r102.id), "(1) обе копии попали в комнату 102");

  const srcUnchanged = JSON.stringify(srcAfter.map(p => ({ x: p.x, y: p.y, frameId: p.frameId, cross: p.cross, mech: p.mech })))
