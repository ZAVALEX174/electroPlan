  // ============ Б4, ч.А: «Отменить/Вернуть» план (Ctrl+Z / Ctrl+Y) ============
  // На входе в хвост: 2 комнаты, пост в немецкой кухне, шаблон post4 в библиотеке. Проверяем:
  // сумма сметы ДО → поставить пост → сумма ВЫРОСЛА → Ctrl+Z (настоящее нажатие через CDP,
  // modifiers:2=Ctrl, code KeyZ) → сумма КАК ДО, пост исчез → Ctrl+Y → снова как после; две стены
  // подряд → две отмены по одной; кнопки disabled в начале (после перезагрузки) и когда отменять
  // нечего; скриншоты холста (клип .canvas-scroll) до/после.
  n = 600;
  const grand = () => ev(`const e=document.getElementById("grandTotal");return e?e.textContent.trim():null`);
  const cntPosts = () => ev(`return document.querySelectorAll(".plan-icon.post").length`);
  const cntWalls = () => ev(`return Array.from(document.querySelectorAll("#wallsSvg line")).filter(l=>l.style.pointerEvents==="none").length`);
  const undoDis = () => ev(`return document.getElementById("undoBtn").disabled`);
  const redoDis = () => ev(`return document.getElementById("redoBtn").disabled`);
  const blur = () => ev(`document.activeElement&&document.activeElement.blur&&document.activeElement.blur();return 1;`);
  const scrollClip = async () => { const r = await rect(`document.querySelector(".canvas-scroll")`); return { x: r.x, y: r.y, width: r.w, height: r.h }; };
  // Настоящее нажатие комбинации через CDP (modifiers:2 = Ctrl). code/vk — физическая клавиша.
  const hotkey = async (code, vk, keyChar) => {
    await c.send("Input.dispatchKeyEvent", { type: "keyDown", modifiers: 2, code, key: keyChar, windowsVirtualKeyCode: vk, nativeVirtualKeyCode: vk });
    await c.send("Input.dispatchKeyEvent", { type: "keyUp",   modifiers: 2, code, key: keyChar, windowsVirtualKeyCode: vk, nativeVirtualKeyCode: vk });
    await sleep(500);
  };
  const ctrlZ = () => hotkey("KeyZ", 90, "z");
  const ctrlY = () => hotkey("KeyY", 89, "y");

  // привести холст к известному состоянию
  await ev(`document.dispatchEvent(new KeyboardEvent("keydown",{key:"Escape",bubbles:true}));return 1;`); await sleep(300);
  await blur();

  const sum0 = await grand(), posts0 = await cntPosts();
  log.push("ДО: сумма=" + sum0 + ", постов=" + posts0 + ", undo.disabled=" + await undoDis() + ", redo.disabled=" + await redoDis());
  await shot("undo-before-panel", [`document.getElementById("undoBtn")`, `document.getElementById("redoBtn")`, `document.querySelector(".grand-total")`]);
  await shot("undo-before-canvas", [], await scrollClip());

  // --- поставить пост (шаблон post4 в немецкую кухню) ---
  await click(`document.querySelector("[data-place-template]")`); await sleep(300);
  const kl = await rect(`Array.from(document.querySelectorAll(".room-label")).find(e=>e.textContent.includes("Кухня"))`);
  await clickAt(kl.x + kl.w / 2, kl.y + kl.h + 45); await sleep(700);
  const sum1 = await grand(), posts1 = await cntPosts();
  log.push("ПОСТАВИЛ: сумма " + sum0 + "→" + sum1 + ", постов " + posts0 + "→" + posts1 + ", undo.disabled=" + await undoDis());
  await shot("undo-after-place-canvas", [], await scrollClip());

  // --- Ctrl+Z: сумма как ДО, пост исчез ---
  await blur();
  await ctrlZ();
  const sum2 = await grand(), posts2 = await cntPosts();
  log.push("Ctrl+Z: сумма " + sum1 + "→" + sum2 + " (ждём " + sum0 + "), постов " + posts1 + "→" + posts2 + " (ждём " + posts0 + ")");
  await shot("undo-after-ctrlz-canvas", [], await scrollClip());

  // --- Ctrl+Y: повторяет отменённое ---
  await ctrlY();
  const sum3 = await grand(), posts3 = await cntPosts();
  log.push("Ctrl+Y: сумма " + sum2 + "→" + sum3 + " (ждём " + sum1 + "), постов " + posts2 + "→" + posts3 + " (ждём " + posts1 + ")");
  await shot("undo-after-ctrly-canvas", [], await scrollClip());

  // --- две стены подряд → две отмены по одной ---
  await click(`document.querySelector('[data-tool="wall"]')`); await sleep(300);
  const a1 = await canvasPt(0.12, 0.12), b1 = await canvasPt(0.24, 0.12), b2 = await canvasPt(0.24, 0.24);
  await clickAt(a1.x, a1.y); await clickAt(b1.x, b1.y);   // стена 1
  await clickAt(b2.x, b2.y);                               // стена 2
  await ev(`document.dispatchEvent(new KeyboardEvent("keydown",{key:"Escape",bubbles:true}));return 1;`); await sleep(300);
  const w0 = await cntWalls();
  log.push("ДВЕ СТЕНЫ: стен стало " + w0);
  await blur();
  await ctrlZ(); const w1 = await cntWalls();
  await ctrlZ(); const w2 = await cntWalls();
  log.push("ДВЕ ОТМЕНЫ стен: " + w0 + "→" + w1 + "→" + w2 + " (каждая отмена снимает одну стену)");

  // --- отменить всё до базы: кнопка «Отменить» гаснет ---
  for (let i = 0; i < 80 && !(await undoDis()); i++) await ctrlZ();
  log.push("ПОСЛЕ ВСЕХ ОТМЕН: undo.disabled=" + await undoDis() + ", redo.disabled=" + await redoDis() + ", сумма=" + await grand() + ", постов=" + await cntPosts());
  await shot("undo-exhausted-panel", [`document.getElementById("undoBtn")`, `document.getElementById("redoBtn")`]);

  // --- перезагрузка: история НЕ переживает F5, обе кнопки disabled в начале ---
  await c.send("Page.reload"); await sleep(4500);
  const uFresh = await ev(`return document.getElementById("undoBtn").disabled`);
  const rFresh = await ev(`return document.getElementById("redoBtn").disabled`);
  log.push("ПОСЛЕ ПЕРЕЗАГРУЗКИ (история не переживает F5): undo.disabled=" + uFresh + ", redo.disabled=" + rFresh);
  await shot("undo-fresh-load-panel", [`document.getElementById("undoBtn")`, `document.getElementById("redoBtn")`]);

  // ============ Б4, ч.Б: ПОДЛОЖКА-ЧЕРТЁЖ в «Отменить/Вернуть» ============
  // Сцена 1: чертёж + пост на холсте → «Убрать план» с подтверждением → чертежа нет → Ctrl+Z →
  //          чертёж и посты на месте (+числа: длина img.src, planLoaded). Сцена 2: загрузка чертежа
  //          настоящим <input> (applyImportedPlan) → F5 → чертёж на месте (сохранился сразу, без правок).
  // Чертёж кладём в УЖЕ существующий проект (с постом) через localStorage + reload — как в rotate.txt.
  n = 700;
  const srcLen = () => ev(`const s=document.getElementById("planImage").src||"";return /^data:/.test(s)?s.length:0`);
  const planLoadedB = () => ev(`return !document.getElementById("clearPlanBtn").hidden`);   // updatePlanUi ведёт hidden по planLoaded
  const chertezBuf = fs.readFileSync(path.join(ROOT, "chertez", "c81e728d9d4c2f636f067f89cc14862c-12.png"));
  const planDataUrl = "data:image/png;base64," + chertezBuf.toString("base64");

  await ev(`const k=ProjectStore.key;const p=JSON.parse(localStorage.getItem(k)||"{}");p.plan=${JSON.stringify(planDataUrl)};p.planLabel="chertez.png";p.planVisibility="show";localStorage.setItem(k,JSON.stringify(p));return 1;`);
  await c.send("Page.reload"); await sleep(5000);
  await ev(`document.getElementById("canvas").scrollIntoView({block:"start"});return 1;`); await sleep(300);

  // ставим пост на холст (шаблон → клик по чертежу): «Убрать план» и его откат не должны его задеть
  await click(`document.querySelector("[data-place-template]")`); await sleep(300);
  const ptB = await canvasPt(0.45, 0.5);
  await clickAt(ptB.x, ptB.y); await sleep(700);

  const lenB0 = await srcLen(), postsB0 = await cntPosts();
  log.push("Б.1 ДО: img.src длина=" + lenB0 + ", planLoaded=" + await planLoadedB() + ", постов=" + postsB0);
  await shot("partB-loaded-canvas", [], await scrollClip());

  // «Убрать план» (задаёт вопрос) → «Точно убрать план?» (подтверждает). Пауза между ними снимает
  // анти-промах «wait» (клик в тот же миг, когда появилась кнопка).
  await click(`document.getElementById("clearPlanBtn")`); await sleep(800);
  await click(`document.getElementById("clearPlanConfirmBtn")`); await sleep(800);
  const lenB1 = await srcLen(), postsB1 = await cntPosts();
  log.push("Б.1 «Убрать план» подтверждён: img.src длина " + lenB0 + "→" + lenB1 + ", planLoaded=" + await planLoadedB() + ", постов " + postsB0 + "→" + postsB1);
  await shot("partB-cleared-canvas", [], await scrollClip());

  // Ctrl+Z → чертёж и посты возвращаются
  await blur(); await ctrlZ();
  const lenB2 = await srcLen(), postsB2 = await cntPosts();
  log.push("Б.1 Ctrl+Z: img.src длина " + lenB1 + "→" + lenB2 + " (ждём " + lenB0 + "), planLoaded=" + await planLoadedB() + ", постов " + postsB1 + "→" + postsB2 + " (ждём " + postsB0 + ")");
  await shot("partB-undo-canvas", [], await scrollClip());

  // Сцена 2: убираем план из localStorage, перезагружаемся (чертежа нет), затем грузим файл настоящим
  // <input type=file> — срабатывает onchange → EPPlanImport → applyImportedPlan → persistProject.
  await ev(`const k=ProjectStore.key;const p=JSON.parse(localStorage.getItem(k)||"{}");delete p.plan;p.planLabel="";localStorage.setItem(k,JSON.stringify(p));return 1;`);
  await c.send("Page.reload"); await sleep(5000);
  log.push("Б.2 перед загрузкой через input: planLoaded=" + await planLoadedB() + ", img.src длина=" + await srcLen());
  const docB = await c.send("DOM.getDocument");
  const inputB = await c.send("DOM.querySelector", { nodeId: docB.root.nodeId, selector: "#planUpload" });
  await c.send("DOM.setFileInputFiles", { files: [path.join(ROOT, "chertez", "c81e728d9d4c2f636f067f89cc14862c-12.png")], nodeId: inputB.nodeId });
  await sleep(5000);   // importFile + applyImportedPlan (+ persistProject внутри onload)
  log.push("Б.2 после загрузки через input (applyImportedPlan): planLoaded=" + await planLoadedB() + ", img.src длина=" + await srcLen());

  // F5 БЕЗ единой ручной правки: чертёж обязан остаться — его сохранил persistProject из applyImportedPlan
  await c.send("Page.reload"); await sleep(5000);
  await ev(`document.getElementById("canvas").scrollIntoView({block:"start"});return 1;`); await sleep(300);
  const lenF5 = await srcLen(), loadedF5 = await planLoadedB();
  log.push("Б.2 ПОСЛЕ F5 (сохранился сразу applyImportedPlan): planLoaded=" + loadedF5 + ", img.src длина=" + lenF5);
  await shot("partB-reload-canvas", [], await scrollClip());

  // ============ Б4 довод (п.1–3, п.6): пересборка комнат, жест переноса, удержание стрелки ============
  // Числа читаем из сохранённого проекта (localStorage, ProjectStore.key): roomLines/rooms/posts.
  n = 800;
  const roomLinesN = () => ev(`return (JSON.parse(localStorage.getItem(ProjectStore.key)||"{}").roomLines||[]).length`);
  const roomsN = () => ev(`return (JSON.parse(localStorage.getItem(ProjectStore.key)||"{}").rooms||[]).length`);
  const postXs = () => ev(`return (JSON.parse(localStorage.getItem(ProjectStore.key)||"{}").posts||[]).map(p=>Math.round(p.x))`);

  // --- п.1: контур разметки 4 клика + пересборка (>600 мс) = 4 шага; 1-й Ctrl+Z снимает последнюю линию ---
  await click(`document.getElementById("clearBtn")`); await sleep(800);   // чистый холст для детерминизма
  await click(`document.querySelector('[data-tool="roomline"]')`); await sleep(300);
  const q1 = await canvasPt(0.30, 0.30), q2 = await canvasPt(0.55, 0.30), q3 = await canvasPt(0.55, 0.55), q4 = await canvasPt(0.30, 0.55);

  await clickAt(q1.x, q1.y);   // замыкание на первую точку → 4-я линия контура
  await sleep(900);            // >600 мс: автопересборка помещений
  log.push("Б4-1 контур нарисован: линий=" + await roomLinesN() + ", комнат=" + await roomsN() + " (ждём линий=4, комнат≥1)");
  await shot("b4-contour-built", [], await scrollClip());
  await blur();
  let contourSteps = 0, firstZlines = null, firstZrooms = null;
  for (let i = 0; i < 10 && (await roomLinesN()) > 0; i++) {
    await ctrlZ(); contourSteps++;
    if (i === 0) { firstZlines = await roomLinesN(); firstZrooms = await roomsN(); }
  }
  log.push("Б4-1 1-й Ctrl+Z: линий=" + firstZlines + " (ждём 3), комнат=" + firstZrooms + " (ждём 0 — комната не осталась призраком)");
  log.push("Б4-1 шагов до снятия всех линий контура = " + contourSteps + " (ждём 4, не 5)");

  // --- п.2: удалить авто-комнату → Z → Y → комнат как после удаления ---
  // заново строим контур (его только что отменили)
  await click(`document.querySelector('[data-tool="roomline"]')`); await sleep(300);
  await clickAt(q1.x, q1.y); await clickAt(q2.x, q2.y); await clickAt(q3.x, q3.y); await clickAt(q4.x, q4.y); await clickAt(q1.x, q1.y);
  await sleep(900);
  const roomsBuilt = await roomsN();
  // удаляем комнату: инструмент «Удалить» + клик по центру контура
  await click(`document.querySelector('[data-tool="delete"]')`); await sleep(300);
  const qc = await canvasPt(0.425, 0.425); await clickAt(qc.x, qc.y); await sleep(800);
  const roomsDeleted = await roomsN();
  await blur(); await ctrlZ(); const roomsAfterZ = await roomsN();
  await ctrlY(); const roomsAfterY = await roomsN();
  log.push("Б4-2 авто-комната: построено=" + roomsBuilt + " → удалено=" + roomsDeleted + " → Ctrl+Z=" + roomsAfterZ + " → Ctrl+Y=" + roomsAfterY + " (ждём Y = как после удаления = " + roomsDeleted + ")");
  await shot("b4-room-delete-redo", [], await scrollClip());

  // --- п.3: поставить пост и сразу перетащить → Ctrl+Z возвращает точку ПОСТАНОВКИ ---
  await click(`document.getElementById("clearBtn")`); await sleep(800);
  await click(`document.querySelector("[data-place-template]")`); await sleep(300);
  const pp = await canvasPt(0.45, 0.5); await clickAt(pp.x, pp.y); await sleep(300);   // поставлен (дебаунс 700 мс взведён)
  const xPlaced = await postXs();
  await drag(pp.x, pp.y, 160, 90);   // сразу тащим (мышь down→move→up), затем пауза
  const xDragged = await postXs();
  await blur(); await ctrlZ(); const xAfterZ = await postXs();
  log.push("Б4-3 пост: поставлен x=" + JSON.stringify(xPlaced) + " → перенесён x=" + JSON.stringify(xDragged) + " → Ctrl+Z x=" + JSON.stringify(xAfterZ) + " (ждём = точка постановки " + JSON.stringify(xPlaced) + ", НЕ промежуточная)");
  await shot("b4-drag-undo", [], await scrollClip());

  // --- п.6: 60 автоповторов зажатой стрелки = ОДИН шаг (Ctrl+Z возвращает исходную точку) ---
  await click(`document.querySelector(".plan-icon.post")`); await sleep(300);   // выделяем пост
  const xBeforeArrow = await postXs();
  for (let i = 0; i < 60; i++) await c.send("Input.dispatchKeyEvent", { type: "rawKeyDown", code: "ArrowRight", key: "ArrowRight", windowsVirtualKeyCode: 39, nativeVirtualKeyCode: 39, autoRepeat: i > 0 });
  await c.send("Input.dispatchKeyEvent", { type: "keyUp", code: "ArrowRight", key: "ArrowRight", windowsVirtualKeyCode: 39, nativeVirtualKeyCode: 39 });
  await sleep(900);
  const xAfterArrow = await postXs();
  await blur(); await ctrlZ(); const xAfterArrowZ = await postXs();
  log.push("Б4-6 удержание стрелки: до=" + JSON.stringify(xBeforeArrow) + " → после 60 автоповторов=" + JSON.stringify(xAfterArrow) + " → 1×Ctrl+Z=" + JSON.stringify(xAfterArrowZ) + " (ждём один Ctrl+Z вернул к " + JSON.stringify(xBeforeArrow) + ")");
  await shot("b4-arrow-hold-undo", [], await scrollClip());
