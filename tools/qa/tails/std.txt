  // ============ В3: строка «Стандарт: …» над «Подобрать накладку» в виде «С картинками» ============
  // Общая часть оставила две комнаты со стандартом (Кухня — немецкий, Гостиная — итальянский). Стираем у всех
  // отделку в сохранённом проекте и перезагружаем: чистая Гостиная — «Стандарт: не задан».
  n = 900;
  await sleep(1500);   // дать автосейву общей части записаться
  await ev(`const k=ProjectStore.key;const p=JSON.parse(localStorage.getItem(k));
    p.rooms.forEach(r=>{delete r.standard;delete r.collection;delete r.frameMaterial;delete r.frameShape;delete r.frameColor;});
    localStorage.setItem(k,JSON.stringify(p));return p.rooms.length;`);
  await c.send("Page.reload"); await sleep(5000);
  await ev(`document.getElementById("canvas").scrollIntoView({block:"start"});return 1;`);

  const roomLabel = `Array.from(document.querySelectorAll(".room-label")).find(e=>e.textContent.includes("Гостиная"))`;
  const stdLine = `document.querySelector(".room-facing-standard")`;
  const lineText = async () => ev(`return (${stdLine})?.textContent||"— строки нет —";`);
  const showView = async (view) => {   // "list" | "pictures": жмём переключатель, только если вид ещё не активен
    const id = view === "list" ? "roomFacingViewList" : "roomFacingViewPictures";
    await ev(`const b=document.getElementById(${JSON.stringify(id)});if(b&&!b.classList.contains("active"))b.click();return 1;`); await sleep(400);
  };
  // блок «Отделка накладки» целиком — крупно, а не всё окно: 10-пиксельные подписи на весь экран не читаются
  const blockClip = async () => ev(`const e=document.querySelector(".room-facing-block");e.scrollIntoView({block:"center"});
    const r=e.getBoundingClientRect();return {x:Math.max(0,r.left+scrollX-14),y:Math.max(0,r.top+scrollY-14),width:r.width+28,height:r.height+28};`);
  const check = (label, got, want) => log.push((got === want ? "ОК     " : "ОШИБКА ") + label + ": «" + got + "»" + (got === want ? "" : " (ждали «" + want + "»)"));

  const lb = await rect(roomLabel); if (!lb) throw new Error("нет таблички Гостиной");
  await clickAt(lb.x + 12, lb.y + lb.h / 2); await sleep(500);

  // --- 1. Вид «С картинками», стандарт не выбран → «не задан» ---
  await showView("pictures");
  check("вид «С картинками», стандарт не выбран", await lineText(), "Стандарт: не задан");
  const cl1 = await blockClip();
  await shot("std-1-pictures-none", [stdLine, `document.getElementById("roomFramePickerBtn")`], cl1);

  // --- 2. Стандарт задаём в виде «Списком» (итальянский) ---
  await showView("list");
  await setVal($("roomStandardSelect"), "IT");
  log.push("список: в селекторе выбрано «" + await ev(`const s=document.getElementById("roomStandardSelect");return s.selectedOptions[0]?.textContent;`) + "»; строки-«Стандарт: …» в виде «Списком» нет: " + (await lineText() === "— строки нет —"));
  const cl2 = await blockClip();
  await shot("std-2-list-italian", [`document.getElementById("roomStandardSelect")`], cl2);

  // --- 3. Снова «С картинками» → «итальянский» ---
  await showView("pictures");
  check("вид «С картинками» после выбора списком", await lineText(), "Стандарт: итальянский");
  const cl3 = await blockClip();
  await shot("std-3-pictures-italian", [stdLine, `document.getElementById("roomFramePickerBtn")`], cl3);

  // --- 4. Мастер: «Стандарт» → немецкий → «Применить» → строка обновилась без переключения видов ---
  await click($("roomFramePickerBtn")); await sleep(700);
  await click(`document.querySelector('#framePickerCrumbs [data-step="0"]')`); await sleep(500);
  await click(`document.querySelector('#framePickerBody .frame-pick-tile[data-value="DE"]')`); await sleep(500);
  await click($("applyFramePicker")); await sleep(700);
  check("после «Применить» мастера (немецкий)", await lineText(), "Стандарт: немецкий");
  const cl4 = await blockClip();
  await shot("std-4-pictures-wizard-german", [stdLine, `document.getElementById("roomFramePickerBtn")`], cl4);
